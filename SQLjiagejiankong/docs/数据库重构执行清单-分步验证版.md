### 3.2 创建新表（只创建，不迁移数据）

```javascript
// scripts/create_new_tables.js
const initSqlJs = require('sql.js');
const fs = require('fs');

async function createNewTables() {
  const SQL = await initSqlJs();
  const db = new SQL.Database(fs.readFileSync('./data/price_monitor_test.db'));

  console.log('=== 创建新表 ===\n');

  // 1. 商品表
  console.log('创建 products 表...');
  db.run(`
    CREATE TABLE IF NOT EXISTS products (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      product_url TEXT NOT NULL UNIQUE,
      platform_product_id TEXT NOT NULL,
      brand TEXT NOT NULL,
      display_name TEXT,
      price_threshold REAL NOT NULL DEFAULT 5.0 CHECK(price_threshold >= 0),
      created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
      updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
    )
  `);
  db.run(`CREATE INDEX IF NOT EXISTS idx_products_brand ON products(brand)`);

  // 2. SKU 表
  console.log('创建 skus 表...');
  db.run(`
    CREATE TABLE IF NOT EXISTS skus (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      product_id INTEGER NOT NULL,
      package_name TEXT NOT NULL,
      refraction TEXT NOT NULL DEFAULT '',
      created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
      UNIQUE(product_id, package_name, refraction),
      FOREIGN KEY (product_id) REFERENCES products(id) ON DELETE CASCADE
    )
  `);
  db.run(`CREATE INDEX IF NOT EXISTS idx_skus_product ON skus(product_id)`);

  // 3. 批次表
  console.log('创建 collection_batches 表...');
  db.run(`
    CREATE TABLE IF NOT EXISTS collection_batches (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      product_id INTEGER NOT NULL,
      imported_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
      collected_at DATETIME,
      status TEXT NOT NULL DEFAULT 'importing' CHECK(status IN ('importing', 'success', 'failed')),
      completeness TEXT NOT NULL DEFAULT 'unknown' CHECK(completeness IN ('unknown', 'complete', 'partial')),
      source_record_count INTEGER,
      inserted_record_count INTEGER,
      error_message TEXT,
      FOREIGN KEY (product_id) REFERENCES products(id) ON DELETE CASCADE
    )
  `);
  db.run(`CREATE INDEX IF NOT EXISTS idx_batches_product ON collection_batches(product_id)`);
  db.run(`CREATE INDEX IF NOT EXISTS idx_batches_imported_at ON collection_batches(imported_at)`);

  // 4. 价格快照表
  console.log('创建 price_snapshots 表...');
  db.run(`
    CREATE TABLE IF NOT EXISTS price_snapshots (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      batch_id INTEGER NOT NULL,
      sku_id INTEGER NOT NULL,
      daily_price TEXT NOT NULL,
      promo_price TEXT,
      UNIQUE(batch_id, sku_id),
      FOREIGN KEY (batch_id) REFERENCES collection_batches(id) ON DELETE CASCADE,
      FOREIGN KEY (sku_id) REFERENCES skus(id) ON DELETE CASCADE
    )
  `);
  db.run(`CREATE INDEX IF NOT EXISTS idx_snapshots_batch ON price_snapshots(batch_id)`);
  db.run(`CREATE INDEX IF NOT EXISTS idx_snapshots_sku ON price_snapshots(sku_id)`);

  console.log('\n=== 验证表结构 ===\n');

  const tables = db.exec(`
    SELECT name FROM sqlite_master 
    WHERE type='table' 
    ORDER BY name
  `);

  console.log('数据库中的表:');
  tables[0].values.forEach(([name]) => console.log('  -', name));

  console.log('\n=== 验证外键约束 ===\n');
  db.run('PRAGMA foreign_keys = ON');
  const fk = db.exec('PRAGMA foreign_keys');
  console.log('外键已开启:', fk[0].values[0][0] === 1);

  // 保存
  fs.writeFileSync('./data/price_monitor_test.db', db.export());
  db.close();

  console.log('\n✅ 新表创建完成');
}

createNewTables().catch(console.error);
```
